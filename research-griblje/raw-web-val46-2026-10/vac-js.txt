function formSubmit(action, names, values) {
    inProgress(true);
    var form = document.createElement("form");
    form.style.display = "none";

        var element = document.createElement("input");
        element.value = values[i];

    form.action = action;

}

var visibilityHandler = (function(){
    var stateKey, eventKey, keys = {
        hidden: "visibilitychange",
        webkitHidden: "webkitvisibilitychange",
        mozHidden: "mozvisibilitychange",
        msHidden: "msvisibilitychange"
    };
    for (stateKey in keys) {
        if (stateKey in document) {
            eventKey = keys[stateKey];
            break;
        }
    }
    return function(c) {
        if (c) document.addEventListener(eventKey, c);
        return !document[stateKey];
    }
})();

function parseDate(date, format, errMsg) {
    var dateIndex = 0;
    var formatIndex = 0;
    
    var tempNumber;
    var tempFormatChar;
    var prevFormatChar = null;
    var day = "";
    var month = "";
    var year = "";
    
    try
    {
        var max = 100;
        while(dateIndex < date.length && formatIndex < format.length) {
            if(max-- <= 0) break;
            tempNumber = date.charAt(dateIndex);
            tempFormatChar = format.charAt(formatIndex);
            
            if(/^\d+$/.test(tempNumber)) {
                // Check which date part we are working on.
                if(tempFormatChar == "d") {
                    day += tempNumber;
                    prevFormatChar = tempFormatChar;
                    dateIndex++;
                    formatIndex++;
                }
                else if(tempFormatChar == "M") {
                    month += tempNumber;
                    prevFormatChar = tempFormatChar;
                    dateIndex++;
                    formatIndex++;
                }
                else if(tempFormatChar == "y") {
                    year += tempNumber;
                    prevFormatChar = tempFormatChar;
                    dateIndex++;
                    formatIndex++;
                }
                else {
                    prevFormatChar = null;
                }
            }
            else {
                //alert("NOT A DIGIT: " + date + "<=>" + format + ", " + tempFormatChar + " ?= " + prevFormatChar + ", di:" + dateIndex + ", fi:" + formatIndex + ", d:" + day + ", m:" + month + ", y:" + year);
                // not a digit
                var foundDigit = false;
                dateIndex++;
                while(dateIndex < date.length) {
                    if(max-- <= 0) break;
                    if(/^\d+$/.test(date.charAt(dateIndex))) {
                        foundDigit = true;
                        break;
                    }
                    dateIndex++;
                }
                if(foundDigit) {
                    // Find next format part.
                    var foundNextFormatPart = false;
                    while(dateIndex < date.length) {
                        if(max-- <= 0) break;
                        tempFormatChar = format.charAt(formatIndex);
                        //alert(date + "<=>" + format + ", " + tempFormatChar + " ?= " + prevFormatChar + ", di:" + dateIndex + ", fi:" + formatIndex + ", d:" + day + ", m:" + month + ", y:" + year);
                        if(tempFormatChar == prevFormatChar) {
                            if(tempFormatChar == "d") {
                                day = "0" + day;
                            }
                            else if(tempFormatChar == "M") {
                                month = "0" + month;
                            }
                            else if(tempFormatChar == "y") {
                                year = "0" + year;
                            }
                        }
                        else if(tempFormatChar == "d" || tempFormatChar == "M" || tempFormatChar == "y") {
                            foundNextFormatPart = true;
                            break;
                        }
                        formatIndex++;
                    }
                    if(!foundNextFormatPart) {
                        // End of format string.
                        break;
                    }
                }
                else {
                    // End of date string.
                    break;
                }
            }
            //alert(date + "<=>" + format + ", di:" + dateIndex + ", fi:" + formatIndex + ", d:" + day + ", m:" + month + ", y:" + year);
        }
    }
    catch(err) {
        if(errMsg != null) {
            dialog(errMsg + " (" + err + ")", null, null, null, null, false, false, true, false);
        }
        return null;
    }
    
    if(day.length < 1 || month.length < 1 || year.length < 1) {
        if(errMsg != null) {
            dialog(errMsg, null, null, null, null, false, false, true, false);
        }
        return null;
    }
    
    if(dateIndex < date.length) {
        // not all parts of string were used so it is not in a proper date format
        if(errMsg != null) {
            dialog(errMsg, null, null, null, null, false, false, true, false);
        }
        return null;
    }
    
    //var jsonDateFormat = "yyyy-MM-dd";
    return year + "-" + month + "-" + day;
}

function dataTableFilteredSortExtraUrlParams(domElement, filter, extraUrlParams) {
    try {
        // hide dialog
        dialogHide();
        
        var target = $(domElement);
        target.blur();
        var targetTable = getParentTable(target);
        
        var entityName = targetTable.attr("data-entityName");
        var path = targetTable.attr("data-path");
        var controllerPath =  path + entityName;
        var columnName = target.attr("data-colName");
        var sortColumn = targetTable.attr("data-sortColumn");
        var pageSize = 10;
        if($("#pagerPageSize").val() != null) {
            pageSize = $("#pagerPageSize").val();
        }
        else {
            pageSize = targetTable.attr("data-pageSize");
        }
        var sortIsAsc = targetTable.attr("data-sortIsAsc") == "true";
        var currentPage = targetTable.attr("data-currentPage");
        var totalPages = targetTable.attr("data-totalPages");
        var searchId = targetTable.attr("data-searchId");
        if(searchId != null) {
            searchId = "&searchId=" + searchId;
        }
        else {
            searchId = "";
        }
        
        // check if columnName is null and try to get it from the parent (th)
        if(columnName == null) {
            target = target.parent().children("[data-colName]");
            columnName = target.attr("data-colName");
        }
        if(columnName == null) {
            columnName = sortColumn;
            target = $("#" + entityName + "_Table a[data-colName='" + columnName + "']")
            sortIsAsc = false;
        }
        
        var onlyWithoutTranslations = false;
        if($('#' + entityName + '_WithoutTranslations:checked').length > 0) {
            onlyWithoutTranslations = true;
        }
        
        var filtered = "";
        var requestType = "GET";
        var dataType = "";
        var contentType = "application/x-www-form-urlencoded; charset=UTF-8";
        if(filter != null) {
            filtered = "Filtered";
            requestType = "POST";
            dataType = "json";
            contentType = "application/json";
        }
        var hrefType = "Ajax";
        if(extraUrlParams == exportParameter) {
            hrefType = "Export";
        }
        var href = controllerPath + filtered + hrefType + "?pageSize=" + pageSize + "&page=1&prevSort=" + sortColumn + "&sort=" + columnName + "&prevDir=" + (sortIsAsc ? "ASC" : "DESC") + "&dir=" + (sortColumn == columnName ? (sortIsAsc ? "DESC" : "ASC") : "ASC") + "&notrans=" + onlyWithoutTranslations + searchId;

        //alert(requestType + ", " + dataType + ", " + contentType + ", " + href);
        inProgress(true);
        
        if(extraUrlParams != exportParameter) {
            $.ajax({ // Send request
                type: requestType,
                url: href + extraUrlParams,
                contentType: contentType,
                data: filter, //'{"filter":"name:weori 3rer er;id:12"}',
                dataType: dataType,
                cache: false,
                headers: {
                    'Accept-Language': 'si_sl'
                },
                success: function(response){
                    if(!response.error) {
                        populatePaginatedTableExtraUrlParams(target, targetTable, response, path, entityName, extraUrlParams);
                        inProgress(false);
                    }
                    else {
                        inProgress(false);
                        dialog(response.errorDescription, null, null, null, null, false, false, true, false);
                    }
                    return false;
                },
                error: handleAjaxError
            });
        }
        else {
            href = href + "&filter=" + encodeURIComponent(filter);
            window.location.href = href;
            inProgress(false);
        }
    }
    catch(err) {
        inProgress(false);
    }
    return false;
}

function dataTableFilteredGoToPageExtraUrlParams(domElement, filter, extraUrlParams, page) {

            return false;
        }

        var path = targetTable.attr("data-path");
        if (!entityName || !path) {
            inProgress(false);
            return false;
        }
        var destinationPage = 1;
        if(page < 1) {
            destinationPage = target.attr("data-destinationPage");
        }
        else {
            destinationPage = page;
        }
        var sortColumn = targetTable.attr("data-sortColumn");
        var pageSize = 10;
        if($("#pagerPageSize").val() != null) {
            pageSize = $("#pagerPageSize").val();
        }
        else {
            pageSize = targetTable.attr("data-pageSize");
        }
        var sortIsAsc = targetTable.attr("data-sortIsAsc") == "true";
        var currentPage = targetTable.attr("data-currentPage");
        var totalPages = targetTable.attr("data-totalPages");
        var searchId = targetTable.attr("data-searchId");
        if(searchId != null) {
            searchId = "&searchId=" + searchId;
        }
        else {
            searchId = "";
        }
        
        var onlyWithoutTranslations = false;
        if($('#' + entityName + '_WithoutTranslations:checked').length > 0) {
            onlyWithoutTranslations = true;
        }

        var requestType = "GET";
        var dataType = "";

        if(filter != null) {
            filtered = "Filtered";
            requestType = "POST";
            dataType = "json";
            contentType = "application/json";

        if(destinationPage == "First") {
            href += "&page=1";
        }
        else if(destinationPage == "Last") {
            href += "&page=" + totalPages;
        }
        else if(destinationPage == "Previous") {
            var previous = parseInt(currentPage) - 1;
            href += "&page=" + previous;
        }
        else if(destinationPage == "Next") {
            var next = parseInt(currentPage) + 1;
            href += "&page=" + next;
        }
        else if(!isNaN(destinationPage)) {
            href += "&page=" + destinationPage
        }
        
        // set the target to the column header A element for calling the populatePaginatedTable function
        target = $('#' + entityName + '_Table a[data-colName="' + sortColumn + '"]');
        
        //alert(href + extraUrlParams);
        
        inProgress(true);
        $.ajax({ // Send request
            type: requestType,
            url: href + extraUrlParams,
            contentType: contentType,
            data: filter, //'{"filter":"name:weori 3rer er;id:12"}',
            dataType: dataType,
            cache: false,
            headers: {
                'Accept-Language': locale
            },
            success: function(response){
                if(!response.error) {
                    if (targetTable.attr('id') !== 'searchResults_Table') {
                        populatePaginatedTableExtraUrlParams(target, targetTable, response, path, entityName, extraUrlParams);
                    } else {
                        // TODO
                        populatePaginatedTableExtraUrlParams(target, targetTable, response, path, entityName, extraUrlParams);
                    }

                    inProgress(false);
                }
                else {
                    inProgress(false);
                    dialog(response.errorDescription, null, null, null, null, false, false, true, false);
                }
                return false;
            },
            error: handleAjaxError
        });
    }
    catch(err) {
        inProgress(false);
    }
    return false;
}

function getParentTable(target) {
    try {
        var fuse = 0;
        while(target.prop("tagName") != "TABLE" && fuse++ < 30) {
            target = target.parent();
        }
    }
    catch(err) {
        dialog(err.message, null, null, null, null, false, false, true, false);
    }
    
    return target;
}

function checkForPaginatedTableItem(itemIdAttributeName, item, paginatedTableItemsDict) {
    var returnItem = null;
    if(itemIdAttributeName != null) {
        // Check if item already exists in the dictionary. This is needed because of the jackson way of deserialization.
        
        var ignore = false;
        var id = item[itemIdAttributeName];
        if(id == null) {
            // looks like the same instance was already handled and only id is provided. 
            id = item;
            ignore = true;
        }
        
        //alert(id);
        //alert(JSON.stringify(item));
        
        // add regional archive to the list because the next occurence of the same instance will be ommited and
        // only id will be provided, without the "regionalArchiveId:". 
        if(!ignore && paginatedTableItemsDict[id] == null) {
            paginatedTableItemsDict[id] = item;
            returnItem = item;
        }
        returnItem = paginatedTableItemsDict[id];
        //alert(JSON.stringify(returnItem));
    }
    
    var returnArray = [];
    returnArray.push(returnItem);
    returnArray.push(paginatedTableItemsDict);
    //alert("inner: " + JSON.stringify(paginatedTableItemsDict));
    
    return returnArray;
}

function populatePaginatedTableExtraUrlParams(target, table, response, path, entityName, extraUrlParams) {
    //alert(JSON.stringify(response));
    //table.attr("data-response", JSON.stringify(response));
    var itemId = table.attr("data-itemId");
    var paginatedTableItemsDict = {};
    
    if(target != null) {
        table.attr("data-sortColumn", response.sortColumn);
        table.attr("data-pageSize", response.pageSize);
        table.attr("data-sortIsAsc", response.sortIsAsc);

                .remove();
            for(i = 1; i <= response.totalPages; i++) {
                $("#pagerSelectPage").append("<option value=" + i + ">" + i + "</option>")
            }
            $("#pagerSelectPage").val(response.currentPage);
            
            $("#pagerTotalPages").text(response.totalPages);

            else {

            }
        }
    }

    var showColumns = table.attr("data-showColumns");
    if(showColumns != null) {
        showColumns = showColumns.split(",");
    }
    var specialColumns = table.attr("data-specialColumns") == "true";
    
    var canEdit = table.attr("data-canEdit") == "true";
    var canDelete = table.attr("data-canDelete") != "false";
    
    if(target != null) {
        $("#" + entityName + "_Table a[data-currentSort='true']").addClass("glyph-sort-updown");
        $("#" + entityName + "_Table a[data-currentSort='true']").removeClass("glyph-sort-down glyph-sort-up");
        $("#" + entityName + "_Table a[data-currentSort='true']").attr("data-currentSort", false);

        target.attr("data-currentSort", true);
        target.removeClass();
        target.addClass("sorting");
        if(response.sortIsAsc) {

        else {

        }

    }
    
    // Load objects from main items
    for (var key in response.pageData){
        paginatedTableItemsDict = checkForPaginatedTableItem(itemId, response.pageData[key], paginatedTableItemsDict)[1];
    }
    // Load objects from translations
    for (var key in response.pageDataTranslations){
        paginatedTableItemsDict = checkForPaginatedTableItem(itemId, response.pageDataTranslations[key], paginatedTableItemsDict)[1];
    }
    
    for (var key = 0, size = response.pageData.length; key < size; key++){
        
        var returnArray = checkForPaginatedTableItem(itemId, response.pageData[key], paginatedTableItemsDict);
        paginatedTableItemsDict = returnArray[1];
        var item = returnArray[0];
        if(item == null) {
            item = response.pageData[key];
        }
        
        var iterLocale = item["locale"];
        var iterLocaleConnectedId = item["localeConnectedId"];
        var showEditButton = canEdit;
        
        if(showLocale && iterLocale != null && iterLocaleConnectedId != null && iterLocale != locale && iterLocale != defaultLocale) {
            continue;
        }
        var tableBodyTr = '<tr '
        var tableBodyTd = '';
        var i = 0;
        var idProperty = null;
        var idValue = -1;
        if(itemId != null) {
            idProperty = itemId;
            idValue = item[itemId];
        }
        
        var deletable = true
        if(showColumns == null) {
            for (var property in item) {
                if(property.charAt(0) == '_') {
                    if(typeof handleSpecialElement === "function") {
                        tableBodyTd += handleSpecialElement(item, property);
                    }
                    continue;
                }
                var value = item[property];
                
                if(typeof value == "boolean") {
                    value = booleanCell(value);
                }
                if(property == "deletable") {
                    deletable = item[property];
                }
                else {
                    if(showLocale) {
                        if(property == "localeConnectedId") {
                            if(value == idValue) {
                                tableBodyTd += '<td style="opacity:0.5" data-property="' + property + '">' + value + '</td>';
                            }

                            }
                        }
                        else {
                            tableBodyTd += '<td data-property="' + property + '">' + value + '</td>';
                        }
                        
                        if(property == "locale" && value != defaultLocale) {
                            showEditButton = false;
                        }
                    }
                    else if(property != "locale" && property != "localeConnectedId") {
                        tableBodyTd += '<td data-property="' + property + '">' + value + '</td>';
                    }
                }
                
                if(idProperty == null && i++ == 0) {
                    idProperty = property;
                    idValue = value;
                }
            }
        }
        else {
            for (var index in showColumns) {
                if(item["deletable"] != null) {
                    deletable = item["deletable"];
                }
                var property = showColumns[index];
                
                var propertyParameters = null;
                if(specialColumns) {
                    //alert("splitting property " + property);
                    propertyParameters = property.split("::");
                    property = propertyParameters[0];
                }
                
                //alert(property);
                //alert(propertyParameters.length);
                var type = null;
                if(property.charAt(0) == '_') {
                    if(typeof handleSpecialElement === "function") {
                        tableBodyTd += handleSpecialElement(item, property);
                    }
                    continue;
                }
                //var value = item[property];
                var value = null;
                if(property.indexOf(".") < 0) {
                    value = item[property];
                }
                else {
                    var propertyParts = property.split(".");
                    value = item;
                    for(var propertyPartIndex in propertyParts) {
                        value = value[propertyParts[propertyPartIndex]];
                    }
                }
                
                if(typeof value == "boolean") {
                    value = booleanCell(value);
                }
                if(showLocale) {
                    if(property == "localeConnectedId") {
                        if(value == idValue) {
                            tableBodyTd += '<td style="opacity:0.5" data-property="' + property + '">' + value + '</td>';
                        }
                        else {
                            tableBodyTd += '<td style="font-weight:bold" data-property="' + property + '">' + value + '</td>';
                        }
                    }
                    else {
                        tableBodyTd += '<td data-property="' + property + '">' + value + '</td>';
                    }
                    
                    if(property == "locale" && value != defaultLocale) {
                        showEditButton = false;
                    }
                }
                else if(property != "locale" && property != "localeConnectedId") {
                    if(propertyParameters == null || propertyParameters.length == 1) {
                        tableBodyTd += '<td data-property="' + property + '">' + (value != null ? value : '') + '</td>';
                    }
                    else {
                        switch(propertyParameters[1]) {
                            case "hidden":
                                tableBodyTd += '<td data-property="' + property + '" style="display:none">' + value + '</td>';
                                break;
                            case "chk":
                                tableBodyTd += '<td>' + checkboxCell(false,value) + '</td>';
                                break;
                            case "boolImgHref":
                                // canBeOrdered::boolImgHref::glyph-shopping-cart::1.5::toCart?id={{id}}<::optional_title>
                                if(item[property]) {
                                    //tableBodyTd += '<td><a href="http://krneki" class="' + propertyParameters[2] + '"></td>';
                                    var ariaLabel = "";
                                    if(propertyParameters.length == 6) {
                                        ariaLabel = " aria-label='" + propertyParameters[5] + "' title='" + propertyParameters[5] + "'";
                                    }
                                    tableBodyTd += '<td><a class="btnInTable ' +propertyParameters[2]+'" style="width:' + propertyParameters[3] + 'rem;font-size:' + propertyParameters[3] +

                                }
                                else {
                                    tableBodyTd += '<td></td>';
                                }
                                break;
                            case "fixTextHref":
                                if(item[property]) {
                                    //tableBodyTd += '<td><a href="http://krneki" class="' + propertyParameters[2] + '"></td>';
                                	//var textValue= /*[[#{UserCart.DeleteCart.Succ}]]*/ null;propertyParameters[5];
                                	tableBodyTd += '<td><a class="" style="width:' + propertyParameters[3] + 'rem;font-size:' + propertyParameters[3] +
                                    'rem;height:' + propertyParameters[3] + 'rem;" href="' + propertyParameters[4].replace("{{id}}", idValue) + '">'+document.getElementById(propertyParameters[5]).innerText+'</a></td>';
                                }
                                else {
                                    tableBodyTd += '<td></td>';
                                }
                                break;
                            case "href":
                                // title::href::details?id={{0}}&refCode={{1}}&type={{2}}::3::id::refCode::SearchProvider
                                var href = propertyParameters[2];
                                for(var i = 0; i < parseInt(propertyParameters[3]); i++) {
                                    href = href.replace("{{" + i + "}}", item[propertyParameters[i + 4]]);
                                }
                                tableBodyTd += '<td><a href="' + encodeURI(href) + '">' + value + '</a></td>';
                                break;
                            case "hrefNewWin":
                                // title::hrefNewWin::details?id={{0}}&refCode={{1}}&type={{2}}::3::id::refCode::SearchProvider
                                if(value != null) {
                                    var href = propertyParameters[2];
                                    for(var i = 0; i < parseInt(propertyParameters[3]); i++) {
                                        href = href.replace("{{" + i + "}}", item[propertyParameters[i + 4]]);
                                    }
                                    if(href.length > 0) {
                                        tableBodyTd += '<td data-property="' + property + '"><a href="#" onclick="window.open(\'' + encodeURI(href) + '\',\'_blank\');return false;">' + value + '</a></td>';
                                    }
                                    else {
                                        tableBodyTd += '<td></td>';
                                    }
                                }
                                else {
                                    tableBodyTd += '<td></td>';
                                }
                                break;
                            case "functionHref":
                                // title::functionHref::action::functionName({{0}}.{{1}}.{{2}});::3::id::refCode::SearchProvider
                                var functionName = propertyParameters[3];
                                functionName = functionName.replace(/\./g, ",");
                                for(var i = 0; i < parseInt(propertyParameters[4]); i++) {
                                    functionName = functionName.replace("{{" + i + "}}", item[propertyParameters[i + 5]]);

                                tableBodyTd += '<td><a href="#" ' + propertyParameters[2] + '="' + functionName + ';return false;">' + value + '</a></td>';

                                if(item[property]) {

                }
            }
        }
        
        // append any custom cells if needed
        if(typeof addCustomCells === "function") {
            tableBodyTd += addCustomCells(response, key);
        }
        
        if(canEdit) {
            var msgEditRow = $("header[data-msgEditRow]").attr("data-msgEditRow");

            if(showEditButton) {

    table.show();

            buildCardsForSearchResults(table);
        }
    } catch (e) {
        console && console.error && console.error(e);
    }
}

/**
 * 
 * @param response
 * @param onClickFunction
 * @param pageNumbersContainer ajax object
 * @param paginationContainer  ajax object selector
 * @returns
 */
function paginationDisplayExtraUrlParams(response, onClickFunction, pageNumbersContainer, paginationContainer, extraUrlParams) {
    pageNumbersContainer.empty();
    pageNumbersContainer.append("&nbsp;&nbsp;&nbsp;");
    
    for(var pageIndex = response.startPage; pageIndex <= response.endPage; pageIndex++) {
        // <a class="pagination-link pagination-row" href="/vac/settings/.regionalArchive.1">1</a>
        var className = "pagination-link pagination-row ";
        if(pageIndex != response.currentPage) {
            className += "pagination-active";
        }
        else {
            className += "pagination-disabled";
        }
        if(extraUrlParams == "") {
            pageNumbersContainer.append("<a class='" + className + "' data-destinationPage='" + pageIndex + "' href='#' onclick='" + onClickFunction + "(this);return false;'>" + pageIndex + "&nbsp;&nbsp;&nbsp;</a>")
        }
        else {
            pageNumbersContainer.append("<a class='" + className + "' data-destinationPage='" + pageIndex + "' href='#' onclick='" + onClickFunction + "ExtraUrlParams(this, \"" + extraUrlParams + "\");return false;'>" + pageIndex + "&nbsp;&nbsp;&nbsp;</a>")
        }
    }

    if(response.currentPage != 1) {
        $(paginationContainer + " a[data-id=PaginationFirst].pagination-disabled").attr("tabindex", 0);
        $(paginationContainer + " a[data-id=PaginationFirst].pagination-disabled").addClass("pagination-active");
        $(paginationContainer + " a[data-id=PaginationFirst].pagination-disabled").removeClass("pagination-disabled");
        $(paginationContainer + " a[data-id=PaginationPrevious].pagination-disabled").attr("tabindex", 0);
        $(paginationContainer + " a[data-id=PaginationPrevious].pagination-disabled").addClass("pagination-active");
        $(paginationContainer + " a[data-id=PaginationPrevious].pagination-disabled").removeClass("pagination-disabled");
    }
    else{
        $(paginationContainer + " a[data-id=PaginationFirst].pagination-active").attr("tabindex", -1);
        $(paginationContainer + " a[data-id=PaginationFirst].pagination-active").addClass("pagination-disabled");
        $(paginationContainer + " a[data-id=PaginationFirst].pagination-active").removeClass("pagination-active");
        $(paginationContainer + " a[data-id=PaginationPrevious].pagination-active").attr("tabindex", -1);
        $(paginationContainer + " a[data-id=PaginationPrevious].pagination-active").addClass("pagination-disabled");
        $(paginationContainer + " a[data-id=PaginationPrevious].pagination-active").removeClass("pagination-active");
    }
    if(response.currentPage != response.totalPages) {
        $(paginationContainer + " a[data-id=PaginationLast].pagination-disabled").attr("tabindex", 0);
        $(paginationContainer + " a[data-id=PaginationLast].pagination-disabled").addClass("pagination-active");
        $(paginationContainer + " a[data-id=PaginationLast].pagination-disabled").removeClass("pagination-disabled");
        $(paginationContainer + " a[data-id=PaginationNext].pagination-disabled").attr("tabindex", 0);
        $(paginationContainer + " a[data-id=PaginationNext].pagination-disabled").addClass("pagination-active");
        $(paginationContainer + " a[data-id=PaginationNext].pagination-disabled").removeClass("pagination-disabled");
    }
    else {
        $(paginationContainer + " a[data-id=PaginationLast].pagination-active").attr("tabindex", -1);
        $(paginationContainer + " a[data-id=PaginationLast].pagination-active").addClass("pagination-disabled");
        $(paginationContainer + " a[data-id=PaginationLast].pagination-active").removeClass("pagination-active");
        $(paginationContainer + " a[data-id=PaginationNext].pagination-active").attr("tabindex", -1);
        $(paginationContainer + " a[data-id=PaginationNext].pagination-active").addClass("pagination-disabled");
        $(paginationContainer + " a[data-id=PaginationNext].pagination-active").removeClass("pagination-active");
    }
}

function changeFontSize(fontSize, byUser){
    //console.log('change font size');
    if(fontSize !== "default") {
        document.documentElement.style.fontSize = fontSize;
    }
    else {
        document.documentElement.style.fontSize = null;
    }
    if(Cookies.get("vac.cookies.allowed")) {
        Cookies.set("vac.accessibility.font.size", fontSize, { expires: 365 });
    }
    
    if(byUser && (isIE11 || isEdge)) {
        // In IE and edge page must be reloaded after the font size is changed.
        location.reload();
    }
    else {
        if(typeof handleFontSizeChange === "function") {
            handleFontSizeChange();
        }
    }
    
    return '';
}

function changeFontFamily(fontFamily, fontWeight){
    //console.log('change font family');
    try {
        if(fontFamily !== "default") {
            document.getElementsByTagName("body")[0].style.fontFamily= fontFamily;
            document.getElementsByTagName("body")[0].style.fontWeight= fontWeight;
            //document.documentElement.style.fontFamily= fontFamily;
            document.getElementById("footer").style.fontFamily= fontFamily;
            document.getElementById("footer").style.fontWeight= fontWeight;
        }
        else {
            document.getElementsByTagName("body")[0].style.fontFamily= null;
            document.getElementsByTagName("body")[0].style.fontWeight= null;
            //document.documentElement.style.fontFamily= fontFamily;
            document.getElementById("footer").style.fontFamily= null;
            document.getElementById("footer").style.fontWeight= null;
        }
    }
    catch(err) {
        //alert(err.message);
    }
    
    if(Cookies.get("vac.cookies.allowed")) {
        Cookies.set("vac.accessibility.font.family", fontFamily, { expires: 365 });

            while (c.charAt(0) == ' ') {
                c = c.substring(1);
            }
            if (c.indexOf(name) == 0) {
                return c.substring(name.length, c.length);
            }
        }
        return "";

function getExportColumns(tableName, propsList) {
    const $table = $("#" + tableName);
    if ($table.length === 0) throw new Error(`Tabela z ID '${tableName}' ne obstaja.`);

    const headers = [];
    $table.find("thead tr.head-hidden th").each(function () {
        const $th = $(this);
        const header =
            ($th.attr("aria-label") || $th.text() || "").trim();
        if (header) headers.push(header);
    });

    const props = Array.isArray(propsList) ? propsList : [];

    return props.map((prop, i) => ({
        prop,
        header: headers[i] || prop
    }));
}

function statusText(code) {
    const $sel = $("#registrationStatus");
    if ($sel.length === 0) return String(code);
    const $opt = $sel.find('option[value="' + code + '"]');
    return $opt.length ? $opt.text() : String(code);
}

function formatCurrentDate(v) {
    if (!v) return "";
    const d = (v instanceof Date) ? v : new Date(v);
    if (isNaN(d)) return String(v);
    // dd.MM.yyyy
    const dd = String(d.getDate()).padStart(2, "0");
    const mm = String(d.getMonth() + 1).padStart(2, "0");
    const yyyy = d.getFullYear();
    return `${dd}.${mm}.${yyyy}`;
}

function findRegionalArchive(value) {

                return tableBodyTd;
            }

        archiveName= value;

    }
    return archiveName;
}

function valueFor(obj, prop) {
    const v = obj[prop];

    switch (prop) {
        case "status":
            return statusText(v);
        case "advancedUser":
        case "hasSicasId":
        case "reproductionOrder":
        case "containsDocuments":
            return yesNo(!!v);
        case "usageStartDate":
        case "inquiryDate":
            return formatCurrentDate(v);
        case "regionalArchive":
            return findRegionalArchive(v);
        case "purpose":
            return findPurpose(v);
        case "loanStatus":
            return findStatusText(v);
        default:
            return v == null ? "" : v
    }
}

function buildRowsForExport(dataArray, columns) {
    return dataArray.map(item => {
        const row = {};
        columns.forEach(col => {
            row[col.header] = valueFor(item, col.prop);
        });
        return row;
    });
}

async function loadTtfAsBase64(url) {
    const res = await fetch(url, { cache: "force-cache" });
    if (!res.ok) throw new Error(`Font fetch failed ${res.status}: ${url}`);
    const buf = await res.arrayBuffer();
    let binary = "";
    const bytes = new Uint8Array(buf);
    const chunk = 0x8000;
    for (let i = 0; i < bytes.length; i += chunk) {
        binary += String.fromCharCode.apply(null, bytes.subarray(i, i + chunk));
    }
    return btoa(binary);
}

function exportUsersToXlsx(dataArray, fileName, selectedProps, tableName) {
    if (!window.XLSX) throw new Error("XLSX library ni naložena.");
    if (!Array.isArray(dataArray)) {
        if (dataArray && Array.isArray(dataArray.pageData)) dataArray = dataArray.pageData;
        else throw new Error("exportUsersToXlsx: expected Array of rows");
    }

    let columns = getExportColumns(tableName, selectedProps);
    if (Array.isArray(selectedProps) && selectedProps.length) {
        const set = new Set(selectedProps);
        columns = columns.filter(c => set.has(c.prop));
    }

    const rows = buildRowsForExport(dataArray, columns);
    const wb = XLSX.utils.book_new();
    const ws = XLSX.utils.json_to_sheet(rows, { header: columns.map(c => c.header) });
    XLSX.utils.book_append_sheet(wb, ws, fileName);

    const headerColor = "1E4760";
    const textColor = "FFFFFF";

    const range = XLSX.utils.decode_range(ws['!ref']);
    const headerRow = range.s.r;

    for (let C = range.s.c; C <= range.e.c; ++C) {
        const cellRef = XLSX.utils.encode_cell({ r: headerRow, c: C });
        const cell = ws[cellRef];
        if (!cell) continue;

        cell.s = {
            fill: {
                patternType: "solid",
                fgColor: { rgb: headerColor }
            },
            font: {
                bold: true,
                color: { rgb: textColor }
            },
            alignment: {
                horizontal: "center",
                vertical: "center"
            }
        };
    }

    XLSX.writeFile(wb, fileName + ".xlsx", { cellStyles: true });
}

async function exportUsersToPdf(dataArray, fileName, selectedProps, tableName) {
    if (!window.jspdf || !window.jspdf.jsPDF) throw new Error("jsPDF UMD ni naložen.");

    const { jsPDF } = window.jspdf;

    if (!Array.isArray(dataArray)) {
        if (dataArray && Array.isArray(dataArray.pageData)) dataArray = dataArray.pageData;
        else throw new Error("exportUsersToPdf: expected Array of rows");
    }

    let columns = getExportColumns(tableName, selectedProps);
    if (!Array.isArray(columns) || columns.length === 0) {
        const first = dataArray[0] || {};
        columns = Object.keys(first).map(k => ({ prop: k, header: k }));
    }

    if (Array.isArray(selectedProps) && selectedProps.length) {
        const set = new Set(selectedProps);
        columns = columns.filter(c => set.has(c.prop));
    }

    const head = [columns.map(c => String(c.header))];
    const body = dataArray.map(item => columns.map(c => valueFor(item, c.prop)));

    const fontUrl = "/fonts/Republika/republika-regular-webfont.ttf";
    const fontBase64 = await loadTtfAsBase64(fontUrl);

    const doc = new jsPDF({ orientation: "landscape", unit: "pt", format: "A4" });
    doc.addFileToVFS("republika-regular-webfont.ttf", fontBase64);
    doc.addFont("republika-regular-webfont.ttf", "Republika", "normal");
    doc.setFont("Republika", "normal");
    doc.setFontSize(14);
    doc.text(fileName, 40, 40);

    doc.autoTable({
        head,
        body,
        startY: 60,
        styles: { fontSize: 9, cellPadding: 4, overflow: "linebreak", font: "Republika" },
        headStyles: { fontStyle: "bold", font: "Republika", fillColor: [30, 71, 96], textColor: 255 },
        margin: { left: 40, right: 40 },
    });

    doc.save(`${fileName}.pdf`);
}
